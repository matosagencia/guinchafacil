private function reboquesOnlineNoRaio(float $lat, float $lng, float $raioKm = 50): int
{
    // Busca guinchos aprovados, disponíveis, com localização
    try {
        $stmt = getPDO()->query(

             FROM guinchos

               AND lat_atual IS NOT NULL AND lng_atual IS NOT NULL"
        );
        $count = 0;
        foreach ($stmt->fetchAll(PDO::FETCH_ASSOC) as $g) {
            $dist = GeoService::haversine($lat, $lng, (float)$g['lat_atual'], (float)$g['lng_atual']);
            $raio = (float)($g['raio_cobertura_km'] ?: $raioKm);
            if ($dist <= $raio) $count++;
        }
        return $count;
    } catch (Throwable $e) {
        error_log('[reboquesOnlineNoRaio] ' . $e->getMessage());
        return 0;
    }
}

Roda no PowerShell:

powershell
cd C:\xampp\htdocs\guinchafacil

$path = ".\src\Controllers\PedidoController.php"
$content = Get-Content $path -Raw -Encoding UTF8
Copy-Item $path "$path.bak-dedup-$(Get-Date -Format 'yyyyMMdd-HHmmss')"
Write-Host "[OK] Backup criado" -ForegroundColor Green

# Conta quantas declarações existem
$matches = [regex]::Matches($content, 'private\s+function\s+nomeEstadoParaUf\s*\(')
Write-Host "[INFO] Ocorrências de nomeEstadoParaUf: $($matches.Count)" -ForegroundColor Cyan

if ($matches.Count -gt 1) {
    # Estratégia: percorre cada declaração a partir da SEGUNDA
    # e remove o bloco completo (do docblock/função até o fechamento balanceado)
    $removidos = 0
    for ($i = $matches.Count - 1; $i -ge 1; $i--) {
        $startIdx = $matches[$i].Index

        # Procura o docblock imediatamente antes (opcional)
        $docStart = $content.LastIndexOf("/**", $startIdx)
        if ($docStart -ge 0 -and ($startIdx - $docStart) -lt 300) {
            $startIdx = $docStart
        }

        # Encontra a abertura '{' após "function nomeEstadoParaUf(...)"
        $braceOpen = $content.IndexOf("{", $matches[$i].Index)
        if ($braceOpen -lt 0) { continue }

        # Conta chaves balanceadas a partir de braceOpen
        $depth = 0
        $braceClose = -1
        for ($j = $braceOpen; $j -lt $content.Length; $j++) {
            $c = $content[$j]
            if ($c -eq '{') { $depth++ }
            elseif ($c -eq '}') {
                $depth--
                if ($depth -eq 0) { $braceClose = $j; break }
            }
        }
        if ($braceClose -lt 0) { continue }

        # Remove do startIdx até braceClose (inclusive) + whitespace em volta
        $endIdx = $braceClose + 1
        while ($endIdx -lt $content.Length -and $content[$endIdx] -match '\s') { $endIdx++ }

        $content = $content.Substring(0, $startIdx) + $content.Substring($endIdx)
        $removidos++
        Write-Host "  [OK] Bloco #$($i+1) removido (linha ~$startIdx)" -ForegroundColor Gray
    }
    Write-Host "[OK] $removidos bloco(s) duplicado(s) removido(s)" -ForegroundColor Green
} else {
    Write-Host "[SKIP] Só existe 1 declaração — sem duplicata" -ForegroundColor Yellow
}

# Salva sem BOM
$utf8NoBom = New-Object System.Text.UTF8Encoding($false)
[System.IO.File]::WriteAllText((Resolve-Path $path).Path, $content, $utf8NoBom)

# Confirma
$final = Get-Content $path -Raw
$finalCount = ([regex]::Matches($final, 'private\s+function\s+nomeEstadoParaUf\s*\(')).Count
Write-Host "[VERIFY] Ocorrências finais: $finalCount" -ForegroundColor Cyan

# Valida sintaxe
& C:\xampp\php\php.exe -l $path
